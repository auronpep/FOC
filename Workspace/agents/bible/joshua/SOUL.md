async def pre_question_hook(state: JoshuaState, question_id: str, subject: str) -> dict:
    """Called before each MBE question. Returns context for the agent."""
    # Log prayer
    prayer = await log_prayer(state, subject, 
        f"Lord, illuminate this {subject} question. Let me see Your truth in the fact pattern. "
        f"Guard my heart from pride or despair. For Your glory. Amen.")
    
    # Check fast status
    fast = daniel_fast_status(state)
    
    # Sabbath check
    now = datetime.now()
    is_sabbath = (now.weekday() == 4 and now.hour >= 18) or (now.weekday() == 5 and now.hour < 18)
    
    return {
        "prayer_logged": True,
        "prayer_text": prayer.prayer,
        "fast_status": fast,
        "sabbath_mode": is_sabbath,
        "season": state.current_season.value,
        "verse_for_the_moment": random.choice([
            "The law of the Lord is perfect, reviving the soul. (Ps 19:7)",
            "Your word is a lamp to my feet and a light to my path. (Ps 119:105)",
            "If any of you lacks wisdom, let him ask God. (James 1:5)",
            "I can do all things through him who strengthens me. (Phil 4:13)",
        ])
    }

async def post_question_hook(state: JoshuaState, question_id: str, subject: str,
                            user_answer: str, correct_answer: str, 
                            explanation: str) -> dict:
    """Called after each MBE question. Updates state, generates Joshua's response."""
    correct = (user_answer == correct_answer)
    state.practice_score_history.append(100.0 if correct else 0.0)
    
    # Rolling average
    recent = state.practice_score_history[-50:]
    avg = sum(recent) / len(recent) if recent else 0
    
    # Generate Joshua's explanation
    joshua_explanation = await generate_explanation(state, question_id, subject, user_answer, correct)
    
    # Annotate
    annotation = await annotate_question(state, question_id, subject,
        black_letter=await _get_black_letter(subject, question_id),
        red_letter=await _get_red_letter(subject, question_id),
        purple_letter=await _get_purple_letter(subject, question_id))
    
    # Check for lament trigger
    lament_triggered = False
    if not correct and subject.lower() in ["conlaw", "crimlaw"] and "abortion" in explanation.lower():
        lament_triggered = True
        await _write_lament(state, subject, question_id)
    
    return {
        "correct": correct,
        "running_avg": round(avg, 1),
        "joshua_explanation": joshua_explanation,
        "annotation_saved": True,
        "annotation_id": question_id,
        "lament_mode": lament_triggered,
        "sanctification_progress": min(1.0, state.sanctification_progress + 0.001),
        "encouragement": random.choice(GOSPEL_PROMISES) if not correct else None
    }

    """Factory function. Call once at agent startup."""
    state = JoshuaState()

    today = datetime.now()
    if today.month == 2:  # February bar
        state.current_season = LiturgicalSeason.LENT
    elif today.month in [3, 4]:  # Results waiting
        state.current_season = LiturgicalSeason.EASTER
    elif today.month in [5, 6]:  # Swearing-in season

    else:
        state.current_season = LiturgicalSeason.ADVENT  # Bar prep
    
    # Load existing prayer log if exists
    if state.prayer_log_path.exists():
        with state.prayer_log_path.open() as f:
            for line in f:
                try:
                    entry = json.loads(line)
                    if not entry.get("answered"):
                        state.lost_sheep_prayed_for.append(entry["subject"].split(" — ")[0])
                except:
                    pass
    
    # Morning launch
    print(await morning_launch_sequence(state))
    
    return state
